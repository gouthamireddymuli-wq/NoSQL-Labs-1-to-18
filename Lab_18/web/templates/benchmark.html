Multi-Database Query Syntax Equivalents

MongoDB (Python/MQL) B-Tree / Aggregation 
# Q1: User History
col.find({"user_id": "U-0001"})\
   .sort("timestamp", -1)\
   .limit(50)

# Q2: High-Risk
col.find({"risk_score": {"$gt": 0.8}})

# Q3: Category Analytics
col.aggregate([
  {"$group": {
    "_id": "$category",
    "total": {"$sum": "$amount"},
    "count": {"$sum": 1}
  }},
  {"$sort": {"total": -1}}
])
Cassandra (CQL) Partition Key 
-- Q1: User History (Optimal)
SELECT * FROM transactions 
WHERE user_id = 'U-0001' 
LIMIT 50;

-- Q2: High-Risk (Inefficient)
CREATE INDEX idx_risk 
  ON transactions(risk_score);
SELECT * FROM transactions 
WHERE risk_score > 0.8 
ALLOW FILTERING;

-- Q3: Analytics (Unsupported)
-- Cassandra has no native $group
-- Requires Apache Spark connector
Neo4j (Cypher) Graph Adjacency 
// Q1: User History
MATCH (u:User {id: 'U-0001'})
  -[:MADE]->(t:Transaction)
RETURN t ORDER BY t.timestamp DESC 
LIMIT 50;

// Q2: Fraud Ring Detection
MATCH (u1:User)-[:SHARES_DEVICE*1..3]-(u2:User)
WHERE u1.id = 'U-0001'
WITH u1, u2, COUNT(*) AS shared
WHERE shared >= 2
RETURN u1.id, u2.id, shared;

// Q3: Analytics (Slow across graph)
MATCH (t:Transaction)
RETURN t.category, SUM(t.amount);
