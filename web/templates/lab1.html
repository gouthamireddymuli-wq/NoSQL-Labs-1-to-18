MongoDB View Definition Pipeline:
db.createView( "employees_safe", // View Name (accessible to regular users) "employees", // Underlying Source Collection (restricted to HR role) [ { "$project": { "emp_id": 1, "name": 1, "department": 1, "role": 1, "performance_rating": 1, // Mask SSN: keep only the last 4 digits "ssn": { "$concat": ["***-**-", { "$substrCP": ["$ssn", 7, 4] }] }, // Redact salary completely from result documents "salary": "$$REMOVE" } } ] )
